import os
import requests
from datetime import datetime
from openai import OpenAI

client = OpenAI(
    base_url="https://models.inference.ai.azure.com",
    api_key=os.environ["GITHUB_TOKEN"],
)

MODEL = "gpt-4o-mini"

def run_agent(role, instruction, context=""):
    response = client.chat.completions.create(
        model=MODEL,
        messages=[
            {"role": "system", "content": f"You are {role}. Provide clear, structured, and actionable output."},
            {"role": "user", "content": f"{instruction}\n\nContext:\n{context}"}
        ]
    )
    return response.choices[0].message.content

topic = os.environ.get("MAIN_TOPIC", "AI business productivity tips")

# AGENT 1: Coordinator
brief = run_agent("Agent 1: Project Manager", f"Create a concise project brief with 3 key objectives for the topic: {topic}")

# AGENT 2: Research & Analysis
research = run_agent("Agent 2: Researcher & Analyst", "Provide supporting facts, current trends, and practical insights based on this brief.", brief)

# AGENT 3: Content Writer
draft = run_agent("Agent 3: Content Writer", "Write a short informative article and an engaging social media caption based on this research.", research)

# AGENT 4: Asset Producer
html_product = run_agent("Agent 4: Digital Product Producer", "Convert the following content into a clean, ready-to-publish HTML page with modern embedded CSS.", draft)

# AGENT 5: Quality Assurance Reviewer
final_caption = run_agent("Agent 5: Quality Assurance Reviewer", "Review and polish the social media caption from this draft so it is error-free and ready to post. Output ONLY the final caption text.", draft)

# AGENT 6: Executor & Marketer
timestamp = datetime.now().strftime("%Y%m%d_%H%M")
os.makedirs("output_products", exist_ok=True)
html_filename = f"output_products/product_{timestamp}.html"
caption_filename = f"output_products/caption_{timestamp}.txt"

with open(html_filename, "w", encoding="utf-8") as f:
    f.write(html_product)

with open(caption_filename, "w", encoding="utf-8") as f:
    f.write(final_caption)

print("Product successfully executed:", html_filename)

webhook_url = os.environ.get("SOCIAL_WEBHOOK")
if webhook_url:
    requests.post(webhook_url, json={"caption": final_caption, "file": html_filename})

name: Run 6-Agent AI Team

on:
  workflow_dispatch:
    inputs:
      topic:
        description: 'Enter today topic or product command'
        required: true
        default: 'Automated digital business ideas'
  schedule:
    - cron: '0 1 * * *'

permissions:
  contents: write
  models: read

jobs:
  execute-team:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Install dependencies
        run: pip install -r requirements.txt

      - name: Run 6-Agent AI Team
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          MAIN_TOPIC: ${{ github.event.inputs.topic || 'Daily business automation tips' }}
          SOCIAL_WEBHOOK: ${{ secrets.SOCIAL_WEBHOOK }}
        run: python tim_ai.py

      - name: Commit and push generated products
        run: |
          git config user.name "AI Executor Bot"
          git config user.email "bot@github.com"
          git add output_products/
          git commit -m "New product generated by AI Team" || exit 0
          git push
